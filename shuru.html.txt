    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // Set year dynamically
            document.getElementById('year').textContent = new Date().getFullYear();

            // 1. Mobile Menu Toggle
            const hamburgerBtn = document.getElementById('hamburger-btn');
            const navMenu = document.getElementById('nav-menu');
            const navLinks = document.querySelectorAll('.nav-link');

            hamburgerBtn.addEventListener('click', () => {
                hamburgerBtn.classList.toggle('active');
                navMenu.classList.toggle('active');
            });

            navLinks.forEach(link => {
                link.addEventListener('click', () => {
                    hamburgerBtn.classList.remove('active');
                    navMenu.classList.remove('active');
                });
            });

            // 2. Scroll Sticky Navbar & Active Anchors
            const navbar = document.getElementById('navbar');
            const sections = document.querySelectorAll('section');

            window.addEventListener('scroll', () => {
                if (window.scrollY > 40) {
                    navbar.classList.add('scrolled');
                } else {
                    navbar.classList.remove('scrolled');
                }

                let current = '';
                sections.forEach(section => {
                    const sectionTop = section.offsetTop - 130;
                    if (window.scrollY >= sectionTop) {
                        current = section.getAttribute('id');
                    }
                });

                navLinks.forEach(link => {
                    link.classList.remove('active');
                    if (link.getAttribute('href') === `#${current}`) {
                        link.classList.add('active');
                    }
                });
            });

            // 3. Animated Counter for Stats Section (500 Subs, 100K Views, 150 Videos)
            const counters = document.querySelectorAll('.counter');
            let animated = false;

            function runCounters() {
                counters.forEach(counter => {
                    const target = +counter.getAttribute('data-target');
                    const duration = 1800; // Total duration ms
                    const step = Math.ceil(target / (duration / 20));
                    let current = 0;

                    const timer = setInterval(() => {
                        current += step;
                        if (current >= target) {
                            counter.textContent = target;
                            clearInterval(timer);
                        } else {
                            counter.textContent = current;
                        }
                    }, 20);
                });
            }

            const statsSection = document.getElementById('stats');
            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting && !animated) {
                        runCounters();
                        animated = true;
                    }
                });
            }, { threshold: 0.3 });

            if (statsSection) {
                observer.observe(statsSection);
            }

            // 4. Dynamic YouTube Video Fetcher via rss2json
            const channelID = 'UCF-FhSCpeSX0_OlX8UlMLbg';
            const rssFeedUrl = `https://www.youtube.com/feeds/videos.xml?channel_id=${channelID}`;
            const apiUrl = `https://api.rss2json.com/v1/api.json?rss_url=${encodeURIComponent(rssFeedUrl)}`;

            const videoFeedContainer = document.getElementById('video-feed');
            const videoModal = document.getElementById('video-modal');
            const modalClose = document.getElementById('modal-close');
            const youtubePlayer = document.getElementById('youtube-player');

            function extractVideoID(guid, link) {
                if (guid && guid.includes('yt:video:')) {
                    return guid.replace('yt:video:', '');
                }
                if (link) {
                    const match = link.match(/(?:v=|\/)([\w-]{11})/);
                    if (match) return match[1];
                }
                return null;
            }

            const fallbackVideos = [
                { title: "SPOILED BACHA GAMING HIGHLIGHTS", id: "dQw4w9WgXcQ", pubDate: "2026-02-15" },
                { title: "NEW GAMING SETUP & UNBOXING", id: "3JZ_D3ELwOQ", pubDate: "2026-02-10" },
                { title: "TOP GAMING MOMENTS OF THE MONTH", id: "L_LUpnjgPso", pubDate: "2026-02-01" }
            ];

            fetch(apiUrl)
                .then(res => {
                    if (!res.ok) throw new Error('API fetch error');
                    return res.json();
                })
                .then(data => {
                    if (data.status === 'ok' && data.items && data.items.length > 0) {
                        renderVideos(data.items.slice(0, 3));
                    } else {
                        renderFallback();
                    }
                })
                .catch(() => renderFallback());

            function renderVideos(items) {
                videoFeedContainer.innerHTML = '';
                items.forEach(item => {
                    const videoId = extractVideoID(item.guid, item.link);
                    if (!videoId) return;

                    const dateStr = item.pubDate ? new Date(item.pubDate).toLocaleDateString() : 'Recent Upload';
                    const thumbnail = `https://img.youtube.com/vi/${videoId}/hqdefault.jpg`;

                    const cardHtml = `
                        <div class="video-card">
                            <div class="thumbnail-container">
                                <img class="thumbnail-img" src="${thumbnail}" alt="${item.title}" loading="lazy" onerror="this.src='https://placehold.co/600x337/121216/8B4513?text=Spoiled+Bacha'">
                                <div class="play-btn-overlay" data-id="${videoId}">
                                    <i class="fa-solid fa-play"></i>
                                </div>
                            </div>
                            <div class="video-content">
                                <h3 class="video-title" title="${item.title}">${item.title}</h3>
                                <div class="video-meta">
                                    <span><i class="fa-regular fa-clock"></i> ${dateStr}</span>
                                    <button class="btn btn-secondary play-trigger" data-id="${videoId}" style="padding: 0.4rem 0.8rem; font-size: 0.75rem;">
                                        Watch Video
                                    </button>
                                </div>
                            </div>
                        </div>
                    `;
                    videoFeedContainer.insertAdjacentHTML('beforeend', cardHtml);
                });

                attachPlayListeners();
            }

            function renderFallback() {
                videoFeedContainer.innerHTML = '';
                fallbackVideos.forEach(item => {
                    const thumbnail = `https://img.youtube.com/vi/${item.id}/hqdefault.jpg`;
                    const cardHtml = `
                        <div class="video-card">
                            <div class="thumbnail-container">
                                <img class="thumbnail-img" src="${thumbnail}" alt="${item.title}" loading="lazy" onerror="this.src='https://placehold.co/600x337/121216/8B4513?text=Spoiled+Bacha'">
                                <div class="play-btn-overlay" data-id="${item.id}">
                                    <i class="fa-solid fa-play"></i>
                                </div>
                            </div>
                            <div class="video-content">
                                <h3 class="video-title">${item.title}</h3>
                                <div class="video-meta">
                                    <span><i class="fa-regular fa-clock"></i> ${item.pubDate}</span>
                                    <button class="btn btn-secondary play-trigger" data-id="${item.id}" style="padding: 0.4rem 0.8rem; font-size: 0.75rem;">
                                        Watch Video
                                    </button>
                                </div>
                            </div>
                        </div>
                    `;
                    videoFeedContainer.insertAdjacentHTML('beforeend', cardHtml);
                });

                attachPlayListeners();
            }

            function attachPlayListeners() {
                document.querySelectorAll('[data-id]').forEach(btn => {
                    btn.addEventListener('click', (e) => {
                        const vid = e.currentTarget.getAttribute('data-id');
                        openModal(vid);
                    });
                });
            }

            function openModal(id) {
                youtubePlayer.src = `https://www.youtube.com/embed/${id}?autoplay=1`;
                videoModal.classList.add('active');
            }

            function closeModal() {
                videoModal.classList.remove('active');
                youtubePlayer.src = '';
            }

            modalClose.addEventListener('click', closeModal);
            videoModal.addEventListener('click', (e) => {
                if (e.target === videoModal) closeModal();
            });

            // 5. Contact Form Submission Toast Handler
            const contactForm = document.getElementById('contact-form');
            const toast = document.getElementById('toast');
            const toastMessage = document.getElementById('toast-message');

            contactForm.addEventListener('submit', (e) => {
                e.preventDefault();
                toastMessage.textContent = 'Message sent! Spoiled Bacha will get back to you soon.';
                toast.classList.add('show');
                contactForm.reset();

                setTimeout(() => {
                    toast.classList.remove('show');
                }, 4000);
            });
        });
    </script>
